cmake_minimum_required(VERSION 3.20)
project(InnoCimguizmo LANGUAGES C CXX)
file(TO_CMAKE_PATH "${INNO_ROOT}" INNO_ROOT)
file(TO_CMAKE_PATH "${INNO_CIMGUI_SOURCE_DIR}" INNO_CIMGUI_SOURCE_DIR)
file(TO_CMAKE_PATH "${INNO_CIMGUI_LIBRARY}" INNO_CIMGUI_LIBRARY)
if(NOT IS_DIRECTORY "${INNO_ROOT}/extern/cimguizmo" OR NOT EXISTS "${INNO_CIMGUI_LIBRARY}")
    message(FATAL_ERROR "The declared cimguizmo sources and published cimgui dependency are required")
endif()
set(CIMGUZMO_ROOT "${INNO_ROOT}/extern/cimguizmo")
add_library(cimguizmo SHARED "${CIMGUZMO_ROOT}/cimguizmo.cpp" "${CIMGUZMO_ROOT}/ImGuizmo/ImGuizmo.cpp")
target_compile_features(cimguizmo PRIVATE cxx_std_11)
target_include_directories(cimguizmo PRIVATE "${CIMGUZMO_ROOT}" "${CIMGUZMO_ROOT}/ImGuizmo"
    "${INNO_CIMGUI_SOURCE_DIR}" "${INNO_CIMGUI_SOURCE_DIR}/imgui")
target_link_libraries(cimguizmo PRIVATE "${INNO_CIMGUI_LIBRARY}")
if(MSVC)
    # Upstream's retained SetID and shared IMGUI_API declarations produce these two known diagnostics.
    target_compile_options(cimguizmo PRIVATE /WX /wd4996 /wd4273)
    target_compile_definitions(cimguizmo PRIVATE "IMGUI_API=__declspec(dllimport)")
    set_target_properties(cimguizmo PROPERTIES OUTPUT_NAME "libcimguizmo")
else()
    target_compile_options(cimguizmo PRIVATE -Werror -Wno-deprecated-declarations)
    set_target_properties(cimguizmo PROPERTIES OUTPUT_NAME "cimguizmo")
endif()
