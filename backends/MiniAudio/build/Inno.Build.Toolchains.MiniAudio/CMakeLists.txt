cmake_minimum_required(VERSION 3.20)
project(InnoMiniAudio LANGUAGES C)
if(NOT IS_DIRECTORY "${INNO_ROOT}/extern/miniaudio")
    message(FATAL_ERROR "INNO_ROOT must identify the explicitly selected checkout")
endif()

    message(FATAL_ERROR "This standalone recipe requires explicit shared linkage")
endif()
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
add_subdirectory("${INNO_ROOT}/extern/miniaudio" upstream)
target_compile_definitions(miniaudio PRIVATE MA_DLL)
