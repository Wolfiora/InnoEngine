using System;
using System.IO;
using System.Collections.Generic;
using System.Diagnostics;
using System.Linq;
using System.Security.Cryptography;
using System.Text;
using System.Xml.Linq;
using System.Text.Json;
using Microsoft.Build.Evaluation;
using Microsoft.Build.Graph;
using Microsoft.Build.Framework;
using Microsoft.Build.Utilities;
using NuGet.Frameworks;

namespace Inno.Build.Toolchains;

/// <summary>
/// Captures an evaluated source graph inside its explicitly selected SDK process.
/// </summary>
public sealed class ManagedInputGraphTask : Task
{
    /// <summary>
    /// Gets or sets the existing project whose graph is evaluated without running build targets.
    /// </summary>
    [Required]
    public string ProjectPath { get; set; } = string.Empty;

    /// <summary>
    /// Gets or sets the operation-owned snapshot destination.
    /// </summary>
    [Required]
    public string OutputPath { get; set; } = string.Empty;

    /// <summary>
    /// Gets or sets the compiler properties borrowed for this graph.
    /// </summary>
    public ITaskItem[] Properties { get; set; } = [];

    /// <summary>
    /// Gets or sets whether declared auxiliary input projects are restored before source capture.
    /// </summary>
    public bool RestoreContributions { get; set; }

    /// <summary>
    /// Gets or sets the operation-owned SDK symbol preparation target.
    /// </summary>
    [Required]
    public string CompilerInputsTargets { get; set; } = string.Empty;

    /// <summary>
    /// Gets or sets project roots for a shared, evaluation-only repository snapshot.
    /// </summary>
    public ITaskItem[] RepositoryProjects { get; set; } = [];

    /// <summary>
    /// Gets or sets whether repository snapshots include SDK-resolved compile references and symbols.
    /// </summary>
    public bool ResolveRepositoryReferences { get; set; }

    /// <summary>
    /// Writes complete source declarations and reports SDK evaluation failures.
    /// </summary>
    /// <returns>
    /// True when the snapshot was written; false when evaluation failed.
    /// </returns>
    public override bool Execute()
    {
        try
        {

            var globals = (Properties ?? new Microsoft.Build.Framework.ITaskItem[0])
                .ToDictionary(item => item.ItemSpec, item => item.GetMetadata("Value"), StringComparer.OrdinalIgnoreCase);
            if (RepositoryProjects.Length != 0)
                return CaptureRepository(globals);
            var result = new XElement("ManagedInputs");
            var paths = new HashSet<string>(OperatingSystem.IsWindows() ? StringComparer.OrdinalIgnoreCase : StringComparer.Ordinal);
            var compilerProperties = new[] {
                "TargetFramework", "TargetFrameworkMoniker", "TargetPlatformMoniker", "TargetPlatformIdentifier", "TargetPlatformVersion",
                "RuntimeIdentifier", "Configuration", "AssemblyName", "OutputType",
                "DefineConstants", "LangVersion", "Nullable", "AllowUnsafeBlocks", "Optimize", "CheckForOverflowUnderflow",
                "PlatformTarget", "DebugType", "DebugSymbols", "Deterministic", "PathMap", "TargetType", "RootNamespace",
                "TreatWarningsAsErrors", "WarningsAsErrors", "WarningsNotAsErrors", "NoWarn", "WarningLevel", "AnalysisLevel",
                "RunAnalyzers", "RunAnalyzersDuringBuild", "EnableNETAnalyzers", "EnforceCodeStyleInBuild", "GenerateDocumentationFile",
                "SignAssembly", "DelaySign", "PublicSign", "AssemblyOriginatorKeyFile", "KeyContainerName", "Features",
                "EnableDefaultCompileItems", "GenerateAssemblyInfo", "GenerateTargetFrameworkAttribute", "ImplicitUsings",
                "EnableDefaultEmbeddedResourceItems", "EmbeddedResourceUseDependentUponConvention", "TargetLatestRuntimePatch",
                "NETCoreSdkVersion", "NetCoreTargetingPackRoot", "TargetingPackVersion", "InnoGeneratePlayerComposition",
                "InnoBgfxShaderTarget", "InnoBgfxShaderApis", "InnoToolTarget", "InnoShaderToolSourceRoot"
            };
            using (var collection = new ProjectCollection())
            {
                var pending = new Queue<ProjectGraph>();
                pending.Enqueue(new ProjectGraph(ProjectPath, globals, collection));
                var visited = new HashSet<string>(StringComparer.Ordinal);
                var restoredContributions = new HashSet<string>(StringComparer.Ordinal);
                while (pending.Count != 0)
                foreach (var node in pending.Dequeue().ProjectNodes.OrderBy(node => node.ProjectInstance.FullPath, StringComparer.Ordinal)
                    .ThenBy(node => string.Join(";", node.ProjectInstance.GlobalProperties.OrderBy(pair => pair.Key, StringComparer.Ordinal)
                        .Select(pair => pair.Key + "=" + pair.Value)), StringComparer.Ordinal))
                {
                    var project = node.ProjectInstance;
                    string identity = project.FullPath + "|" + string.Join(";", project.GlobalProperties.OrderBy(pair => pair.Key, StringComparer.Ordinal)
                        .Select(pair => pair.Key + "=" + pair.Value));
                    if (!visited.Add(identity))
                        continue;
                    foreach (var contribution in project.GetItems("InnoBuildInputProject"))
                    {
                        var selected = new Dictionary<string, string>(project.GlobalProperties, StringComparer.OrdinalIgnoreCase);
                        foreach (string name in contribution.GetMetadataValue("GlobalPropertiesToRemove").Split(';', StringSplitOptions.RemoveEmptyEntries))
                            selected.Remove(name);
                        string dependency = Path.GetFullPath(contribution.EvaluatedInclude, Path.GetDirectoryName(project.FullPath));
                        string restoreKey = dependency + "|" + string.Join(";", selected
                            .Where(pair => !pair.Key.Equals("ArtifactsPath", StringComparison.OrdinalIgnoreCase))
                            .OrderBy(pair => pair.Key, StringComparer.Ordinal)
                            .Select(pair => pair.Key + "=" + pair.Value));
                        // Distinct graph properties must not overwrite a shared dependency's restored RID targets.
                        string layout = globals["ArtifactsPath"];
                        selected["ArtifactsPath"] = Path.Combine(layout, "contributions",
                            Convert.ToHexStringLower(SHA256.HashData(Encoding.UTF8.GetBytes(restoreKey))));
                        if (RestoreContributions && restoredContributions.Add(restoreKey))
                        {
                            var restoreProperties = new System.Collections.Hashtable(StringComparer.OrdinalIgnoreCase);
                            foreach (var property in selected)
                                restoreProperties[property.Key] = property.Value;
                            if (!BuildEngine.BuildProjectFile(dependency, new[] { "Restore" }, restoreProperties, null))
                                throw new InvalidOperationException("Declared input contribution restore failed: " + dependency);
                        }
                        pending.Enqueue(new ProjectGraph(dependency, selected, collection));
                    }
                    if (RestoreContributions)
                        continue;
                    string root = Path.GetDirectoryName(project.FullPath);
                    paths.Add(project.FullPath);
                    foreach (string import in project.ImportPaths)
                        paths.Add(import);
                    foreach (var property in project.GlobalProperties.OrderBy(pair => pair.Key, StringComparer.Ordinal))
                        result.Add(new XElement("Declaration", project.FullPath + "|global:" + property.Key + "=" + property.Value));
                    foreach (string name in compilerProperties)
                        result.Add(new XElement("Declaration", project.FullPath + "|" + name + "=" + project.GetPropertyValue(name)));
                    if (project.Targets.ContainsKey("AddImplicitDefineConstants"))
                    {
                        var symbolProperties = new System.Collections.Hashtable(StringComparer.OrdinalIgnoreCase);
                        foreach (var property in project.GlobalProperties)
                            symbolProperties[property.Key] = property.Value;
                        symbolProperties["InnoOriginalCustomAfterMicrosoftCommonTargets"] = project.GetPropertyValue("CustomAfterMicrosoftCommonTargets");
                        symbolProperties["CustomAfterMicrosoftCommonTargets"] = CompilerInputsTargets;
                        var outputs = new System.Collections.Hashtable(StringComparer.OrdinalIgnoreCase);
                        if (!BuildEngine.BuildProjectFile(project.FullPath, new[] { "CaptureInnoCompilerDefines" }, symbolProperties, outputs))
                            throw new InvalidOperationException("SDK compiler symbol preparation failed: " + project.FullPath);
                        var symbols = (ITaskItem[])outputs["CaptureInnoCompilerDefines"];
                        result.Add(new XElement("Declaration", project.FullPath + "|CompilerDefineConstants="
                            + string.Join(";", symbols.Select(item => item.ItemSpec).Distinct(StringComparer.Ordinal).OrderBy(value => value, StringComparer.Ordinal))));
                    }
                    foreach (var item in project.Items)
                    {
                        bool source = new[] { "Compile", "EmbeddedResource", "Analyzer", "AdditionalFiles", "EditorConfigFiles",
                            "GlobalAnalyzerConfigFiles", "AdditionalCompileInputs" }.Contains(item.ItemType);
                        bool copy = new[] { "Always", "PreserveNewest" }.Contains(item.GetMetadataValue("CopyToOutputDirectory"));
                        bool reference = item.ItemType == "Reference";
                        bool policy = new[] { "PackageReference", "ProjectReference", "FrameworkReference", "CompilerVisibleProperty", "CompilerVisibleItemMetadata", "InnoBuildInputProject" }.Contains(item.ItemType);
                        if (!source && !copy && !reference && !policy)
                            continue;
                        result.Add(new XElement("Declaration", project.FullPath + "|item:" + item.ItemType + "=" + item.EvaluatedInclude));
                        if (item.ItemType == "CompilerVisibleProperty")
                            result.Add(new XElement("Declaration", project.FullPath + "|compiler-visible:" + item.EvaluatedInclude + "=" + project.GetPropertyValue(item.EvaluatedInclude)));
                        foreach (var metadata in item.Metadata.OrderBy(value => value.Name, StringComparer.Ordinal))
                            result.Add(new XElement("Declaration", project.FullPath + "|metadata:" + item.ItemType + ":" + item.EvaluatedInclude
                                + ":" + metadata.Name + "=" + metadata.EvaluatedValue));
                        string file = reference ? item.GetMetadataValue("HintPath") : item.EvaluatedInclude;
                        if (!source && !copy && !reference || string.IsNullOrEmpty(file))
                            continue;
                        file = Path.GetFullPath(file, root);
                        // Generated native bindings have a separate selected-generation input contract.
                        if (item.ItemType == "Compile" && project.GetPropertyValue("BindGenGeneratedBindings") == "true"
                            && project.GetPropertyValue("InnoNativeTarget").Length != 0
                            && file == Path.Combine(root, "Generated", "Bindings.cs"))
                            continue;
                        if (!File.Exists(file))
                        {
                            // The SDK filters nonexistent ancestor configuration candidates before compilation.
                            if (item.ItemType == "EditorConfigFiles" || item.ItemType == "GlobalAnalyzerConfigFiles")
                            {
                                result.Add(new XElement("Declaration", project.FullPath + "|absent-analyzer-config=" + file));
                                continue;
                            }
                            throw new FileNotFoundException("Declared managed " + item.ItemType + " input is absent: " + file, file);
                        }
                        paths.Add(file);
                    }
                    foreach (var framework in project.GetItems("FrameworkReference"))
                    {
                        var known = project.GetItems("KnownFrameworkReference").SingleOrDefault(item => item.EvaluatedInclude == framework.EvaluatedInclude
                            && NuGetFramework.Parse(item.GetMetadataValue("TargetFramework")).DotNetFrameworkName
                                == project.GetPropertyValue("TargetFrameworkMoniker"));
                        if (known == null)
                            throw new InvalidOperationException("Framework reference '" + framework.EvaluatedInclude
                                + "' has no SDK targeting declaration for '" + project.GetPropertyValue("TargetFrameworkMoniker")
                                + "' in '" + project.FullPath + "'.");
                        string version = framework.GetMetadataValue("TargetingPackVersion");
                        if (version.Length == 0)
                            version = known.GetMetadataValue("TargetingPackVersion");
                        string directory = Path.Combine(project.GetPropertyValue("NetCoreTargetingPackRoot"), known.GetMetadataValue("TargetingPackName"), version);
                        if (!Directory.Exists(directory))
                            throw new DirectoryNotFoundException("The selected SDK reference pack is absent: " + directory);
                        foreach (string file in Directory.EnumerateFiles(directory, "*", SearchOption.AllDirectories))
                            paths.Add(file);
                    }
                    // Restored dependency resolution is itself an input; package files are collected by the reader.
                    string assets = project.GetPropertyValue("ProjectAssetsFile");
                    if (assets.Length != 0)
                    {
                        assets = Path.GetFullPath(assets, root);
                        string runtime = project.GetPropertyValue("RuntimeIdentifier");
                        using var restored = JsonDocument.Parse(File.ReadAllText(assets));
                        string alias = project.GetPropertyValue("TargetFramework");
                        string restoredFramework = restored.RootElement.GetProperty("project").GetProperty("frameworks")
                            .EnumerateObject().Where(framework => framework.Value.GetProperty("targetAlias").GetString() == alias)
                            .Select(framework => framework.Name).Single();
                        var selectedFramework = NuGetFramework.Parse(restoredFramework);
                        string selectedTarget = restored.RootElement.GetProperty("targets").EnumerateObject()
                            .Select(target => target.Name).SingleOrDefault(target =>
                            {
                                int separator = target.IndexOf('/');
                                string framework = separator < 0 ? target : target.Substring(0, separator);
                                string targetRuntime = separator < 0 ? string.Empty : target.Substring(separator + 1);
                                return targetRuntime == runtime && NuGetFramework.Parse(framework).Equals(selectedFramework);
                            });
                        if (selectedTarget == null)
                            throw new InvalidDataException("Restored graph '" + assets + "' has no target for '"
                                + selectedFramework.GetShortFolderName() + "' with runtime '" + runtime + "'.");
                        result.Add(new XElement("Assets", new XAttribute("Path", assets), new XAttribute("Target", selectedTarget)));
                    }
                    for (var parent = new DirectoryInfo(root); parent != null; parent = parent.Parent)
                    {
                        string sdkSelection = Path.Combine(parent.FullName, "global.json");
                        if (File.Exists(sdkSelection))
                        {
                            paths.Add(sdkSelection);
                            break;
                        }
                    }
                }
            }
            foreach (string path in paths.OrderBy(path => path, StringComparer.Ordinal))
                result.Add(new XElement("Input", new XAttribute("Path", path)));
            result.Save(OutputPath);
            return true;
        }
        catch (Exception failure)
        {
            Log.LogErrorFromException(failure, true);
            return false;
        }
    }
    private bool CaptureRepository(Dictionary<string, string> globals)
    {
        string[] propertyNames = ["AssemblyName", "OutputType", "InnoProductId", "InnoProductTarget", "InnoNativeTarget",
            "ImplicitUsings", "TargetPath", "IsTestProject", "IsRoslynComponent", "TargetFramework", "RuntimeIdentifier",
            "InnoTestNativeProduct", "InnoTestNativeTarget", "InnoValidationLayer", "InnoValidationEnvironment",
            "InnoValidationIsolation", "InnoValidationReusable", "NETCoreSdkVersion", "NETCoreSdkRuntimeIdentifier", "ProjectAssetsFile",
            "InnoValidationBackendPattern",
            "DefineConstants", "LangVersion", "Nullable", "AllowUnsafeBlocks", "BindGenGeneratedBindings",
            "InnoDevelopmentTarget", "InnoDevelopmentEditorBuildProperties", "InnoDevelopmentEditorProject"];
        string[] itemNames = ["ProjectReference", "Compile", "Using", "EmbeddedResource", "Content", "None", "Analyzer",
            "AdditionalFiles", "EditorConfigFiles", "AdditionalCompileInputs", "InnoBuildInputProject", "InnoValidationInput"];
        var snapshots = new List<object>();
        using var collection = new ProjectCollection(globals);
        var elapsed = Stopwatch.StartNew();
        var graph = new ProjectGraph(RepositoryProjects.Select(item => new ProjectGraphEntryPoint(item.ItemSpec, globals)), collection,
            null, Math.Min(Environment.ProcessorCount, 4), System.Threading.CancellationToken.None);
        Log.LogMessage(MessageImportance.High, $"INNO-SDK-GRAPH evaluationSeconds={elapsed.Elapsed.TotalSeconds:F3} nodes={graph.ProjectNodes.Count}");
        elapsed.Restart();
        var nodes = graph.ProjectNodes.GroupBy(node => node.ProjectInstance.FullPath,
            OperatingSystem.IsWindows() ? StringComparer.OrdinalIgnoreCase : StringComparer.Ordinal)
            .ToDictionary(group => group.Key, group => group.ToArray(),
                OperatingSystem.IsWindows() ? StringComparer.OrdinalIgnoreCase : StringComparer.Ordinal);
        var variants = graph.ProjectNodes.GroupBy(node => node.ProjectInstance.FullPath,
            OperatingSystem.IsWindows() ? StringComparer.OrdinalIgnoreCase : StringComparer.Ordinal)
            .ToDictionary(group => group.Key, group => group.Select(node => node.ProjectInstance).ToArray(),
                OperatingSystem.IsWindows() ? StringComparer.OrdinalIgnoreCase : StringComparer.Ordinal);
        foreach (ITaskItem root in RepositoryProjects.OrderBy(item => item.ItemSpec, StringComparer.Ordinal))
        {
            var project = collection.LoadProject(root.ItemSpec);
            var items = new Dictionary<string, object>(StringComparer.Ordinal);
            var properties = propertyNames.ToDictionary(name => name, name => project.GetPropertyValue(name), StringComparer.Ordinal);
            foreach (string name in itemNames)
                items[name] = project.GetItems(name).Select(item =>
                {
                    var metadata = item.Metadata.OrderBy(value => value.Name, StringComparer.Ordinal)
                        .ToDictionary(value => value.Name, value => value.EvaluatedValue, StringComparer.OrdinalIgnoreCase);
                    metadata["Identity"] = item.EvaluatedInclude;
                    metadata["FullPath"] = item.GetMetadataValue("FullPath");
                    return metadata;
                }).ToArray();
            if (ResolveRepositoryReferences)
            {
                var selected = new System.Collections.Hashtable(StringComparer.OrdinalIgnoreCase);
                foreach (var property in globals)
                    selected[property.Key] = property.Value;
                selected["InnoOriginalCustomAfterMicrosoftCommonTargets"] = project.GetPropertyValue("CustomAfterMicrosoftCommonTargets");
                selected["CustomAfterMicrosoftCommonTargets"] = CompilerInputsTargets;
                var outputs = new System.Collections.Hashtable(StringComparer.OrdinalIgnoreCase);
                if (!BuildEngine.BuildProjectFile(project.FullPath, new[] { "CaptureInnoRepositoryReferences" }, selected, outputs))
                    throw new InvalidOperationException("SDK repository reference resolution failed: " + project.FullPath);
                var captured = (ITaskItem[])outputs["CaptureInnoRepositoryReferences"];
                foreach (string name in new[] { "ReferencePath", "Compile" })
                    items[name] = captured.Where(item => item.GetMetadata("CaptureKind") == name).Select(item =>
                    {
                        var metadata = item.MetadataNames.Cast<string>().ToDictionary(key => key, item.GetMetadata, StringComparer.OrdinalIgnoreCase);
                        metadata["Identity"] = item.ItemSpec;
                        metadata["FullPath"] = item.GetMetadata("FullPath");
                        return metadata;
                    }).ToArray();
                foreach (ITaskItem item in captured.Where(item => item.GetMetadata("CaptureKind") == "Property"))
                    properties[item.ItemSpec] = item.GetMetadata("Value");
            }
            string[] imports = project.Imports.Select(import => import.ImportedProject.FullPath).Distinct().OrderBy(path => path, StringComparer.Ordinal).ToArray();
            string canonical = SnapshotIdentity(properties, items, imports);
            var effective = variants[project.FullPath].OrderBy(instance => string.Join(";", instance.GlobalProperties.OrderBy(pair => pair.Key, StringComparer.Ordinal)
                    .Select(pair => pair.Key + "=" + pair.Value)), StringComparer.Ordinal).Select(instance => new
                {
                    Path = instance.FullPath,
                    GlobalProperties = instance.GlobalProperties,
                    Properties = propertyNames.ToDictionary(name => name, instance.GetPropertyValue, StringComparer.Ordinal),
                    Items = itemNames.ToDictionary(name => name, name => instance.GetItems(name).Select(item =>
                    {
                        var metadata = item.Metadata.OrderBy(value => value.Name, StringComparer.Ordinal)
                            .ToDictionary(value => value.Name, value => value.EvaluatedValue, StringComparer.OrdinalIgnoreCase);
                        metadata["Identity"] = item.EvaluatedInclude;
                        metadata["FullPath"] = item.GetMetadataValue("FullPath");
                        return metadata;
                    }).ToArray(), StringComparer.Ordinal),
                    Imports = instance.ImportPaths.Distinct().OrderBy(path => path, StringComparer.Ordinal).ToArray()
                }).GroupBy(value => SnapshotIdentity(value.Properties, value.Items, value.Imports), StringComparer.Ordinal).ToArray();
            snapshots.Add(new
            {
                Path = project.FullPath,
                GlobalProperties = globals,
                Instances = nodes[project.FullPath].Select(node => new
                {
                    Identity = InstanceIdentity(node),
                    node.ProjectInstance.GlobalProperties,
                    TargetPath = node.ProjectInstance.GetPropertyValue("TargetPath"),
                    ProjectAssetsFile = node.ProjectInstance.GetPropertyValue("ProjectAssetsFile"),
                    References = node.ProjectReferences.Select(InstanceIdentity).OrderBy(value => value, StringComparer.Ordinal).ToArray()
                }).OrderBy(instance => instance.Identity, StringComparer.Ordinal).ToArray(),
                EquivalentGlobalProperties = effective.Where(group => group.Key == canonical).SelectMany(group => group.Select(value => value.GlobalProperties)).ToArray(),
                Properties = properties,
                Items = items,
                Imports = imports,
                // Keep every effective input and every originating property set, without repeating identical item arrays.
                Variants = effective.Where(group => group.Key != canonical).Select(group => new
                {
                    group.First().Path,
                    group.First().GlobalProperties,
                    EquivalentGlobalProperties = group.Select(value => value.GlobalProperties).ToArray(),
                    group.First().Properties,
                    group.First().Items,
                    group.First().Imports
                }).ToArray()
            });
        }
        File.WriteAllText(OutputPath, JsonSerializer.Serialize(snapshots));
        Log.LogMessage(MessageImportance.High, $"INNO-SDK-GRAPH captureSeconds={elapsed.Elapsed.TotalSeconds:F3} bytes={new FileInfo(OutputPath).Length}");
        return true;

        static string InstanceIdentity(ProjectGraphNode node)
            => Convert.ToHexStringLower(SHA256.HashData(JsonSerializer.SerializeToUtf8Bytes(new
            {
                node.ProjectInstance.FullPath,
                Properties = node.ProjectInstance.GlobalProperties.OrderBy(pair => pair.Key, StringComparer.Ordinal).ToArray()
            })));

        static string SnapshotIdentity(
            object properties,
            object items,
            string[] imports
        ) => Convert.ToHexStringLower(SHA256.HashData(JsonSerializer.SerializeToUtf8Bytes(new { Properties = properties, Items = items, Imports = imports })));
    }

}
